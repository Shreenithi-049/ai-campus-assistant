/**
 * FIRESTORE SECURITY RULES
 * 
 * Add these rules to Firebase Console → Firestore Database → Rules
 * 
 * These rules ensure:
 * - Only authenticated users can register
 * - Users can only register with their own userId
 * - Users can read their own registrations
 * - No one can update or delete registrations (admin only via console)
 */

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Students collection - existing rules
    match /students/{userId} {
      allow read, update: if request.auth.uid == userId;
      allow create: if request.auth != null && 
                      request.resource.data.email.matches('.*@skcet\\.ac\\.in$');
    }
    
    // Events collection - read-only for students
    match /events/{eventId} {
      allow read: if request.auth != null;
      allow write: if false; // Admin only (future implementation)
    }
    
    // Event Registrations - NEW RULES
    match /eventRegistrations/{docId} {
      // Allow authenticated users to read their own registrations
      allow read: if request.auth != null && 
                    resource.data.userId == request.auth.uid;
      
      // Allow authenticated users to create registrations with their own userId
      allow create: if request.auth != null && 
                      request.auth.uid == request.resource.data.userId &&
                      request.resource.data.status == 'registered';
      
      // Prevent updates and deletes (admin only via console)
      allow update, delete: if false;
    }
    
    // Faculty collection - read-only
    match /faculty/{facultyId} {
      allow read: if request.auth != null;
      allow write: if false;
    }
    
    // Timetables - read-only
    match /timetables/{document=**} {
      allow read: if request.auth != null;
      allow write: if false;
    }
    
    // Announcements - read-only
    match /announcements/{announcementId} {
      allow read: if request.auth != null;
      allow write: if false;
    }
  }
}
